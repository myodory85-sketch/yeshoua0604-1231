📥 데이터 가져오기
다른 기기에서 내보낸 JSON 파일을 불러오거나,
구글 시트에 저장된 데이터를 이 기기로 가져옵니다.
⚙️ 구글 시트 연동 설정
📌 연동 방법
① Google Sheets에서 새 시트 생성
② 도구 → Apps Script → 코드 붙여넣기
③ 배포 → 웹앱 배포 → URL 복사
④ 아래 URL 입력 후 저장
🔗 앱 공유하기
이 HTML 파일을 아래 방법으로 공유하면 누구나 같은 구글 시트에 함께 저장할 수 있습니다.
📤 공유 방법 3가지
① 카카오톡/메일 — 이 HTML 파일을 직접 첨부 전송
② 구글 드라이브 — 드라이브에 업로드 후 링크 공유
③ GitHub Pages — 무료 웹호스팅으로 URL 링크 공유
⚠️ 공유 전 반드시 URL을 저장한 후 파일을 공유해야 다른 사람도 같은 시트에 저장됩니다.
📋 Apps Script 코드
function doPost(e){ var ss=SpreadsheetApp.getActiveSpreadsheet(); var data=JSON.parse(e.postData.contents); var tab=ss.getSheetByName(data.type)||ss.insertSheet(data.type); if(tab.getLastRow()===0) tab.appendRow(Object.keys(data.payload)); tab.appendRow(Object.values(data.payload)); return ContentService.createTextOutput( JSON.stringify({status:'ok'}) ).setMimeType(ContentService.MimeType.JSON); } function doGet(e){ var ss=SpreadsheetApp.getActiveSpreadsheet(); var type=e.parameter.type||'체크리스트'; var tab=ss.getSheetByName(type); if(!tab) return ContentService.createTextOutput( JSON.stringify({data:[]}) ).setMimeType(ContentService.MimeType.JSON); var rows=tab.getDataRange().getValues(); var h=rows[0]; var result=rows.slice(1).map(function(r){ var o={};h.forEach(function(k,i){o[k]=r[i];});return o; }); return ContentService.createTextOutput( JSON.stringify({data:result}) ).setMimeType(ContentService.MimeType.JSON); }
